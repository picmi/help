# PICMI at a glance

PICMI is an opportunity and engagement platform for organisations. It supports employment, services, placements, and
workforce participation by helping you manage opportunities, intake, applications, agreements, and the move into ongoing
participation.

Use these one-page explainers to see how PICMI fits together. Start with the small previews to understand the whole
product, then open any one-pager when you want to read the detail or download it for sharing.

<script setup lang="ts">
import { onBeforeUnmount, onMounted } from 'vue'
import engagingOrganisations from '../../img/engaging-organisations.png'
import jobseekerSignIn from '../../img/jobseeker-sign-in.png'
import managingApplications from '../../img/managing-applications.png'
import provisionalUsers from '../../img/provisional-users.png'
import recruitingPeople from '../../img/recruiting-people.png'
import understandingNotifications from '../../img/understanding-notifications.png'
import understandingProgressions from '../../img/understanding-progressions.png'
import understandingUsers from '../../img/understanding-users.png'
import upcomingSeason from '../../img/upcoming-season.png'
import workflowDesignHierarchy from '../../img/workflow-design-hierarchy.png'

let cards: NodeListOf<HTMLDetailsElement>

const setHash = (hash: string) => {
  const url = new URL(window.location.href)
  url.hash = hash
  window.history.replaceState(null, '', url)
}

const openCardFromHash = () => {
  const id = window.location.hash.slice(1)
  if (!id) return

  const card = document.getElementById(id)
  if (!(card instanceof HTMLDetailsElement)) return

  cards.forEach((otherCard) => {
    if (otherCard !== card) otherCard.open = false
  })
  card.open = true
  requestAnimationFrame(() => card.scrollIntoView({ block: 'start' }))
}

const updateHashFromCard = (event: Event) => {
  const card = event.currentTarget
  if (!(card instanceof HTMLDetailsElement)) return

  if (card.open) {
    setHash(card.id)
  } else if (window.location.hash === `#${card.id}`) {
    setHash('')
  }
}

onMounted(() => {
  cards = document.querySelectorAll<HTMLDetailsElement>('.one-pager-card')
  cards.forEach((card) => card.addEventListener('toggle', updateHashFromCard))
  window.addEventListener('hashchange', openCardFromHash)
  openCardFromHash()
})

onBeforeUnmount(() => {
  cards?.forEach((card) => card.removeEventListener('toggle', updateHashFromCard))
  window.removeEventListener('hashchange', openCardFromHash)
})
</script>
